import pandas as pd
import numpy as np
from sklearn.model_selection import StratifiedKFold
from sklearn.metrics import accuracy_score, f1_score, classification_report
from xgboost import XGBClassifier
from imblearn.over_sampling import SMOTE

# 1️⃣ Load dataset
df = pd.read_csv('sovereign_default_dataset_1980_2022.csv - sovereign_default_dataset_1980_2022.csv')

# 2️⃣ Define features
features = [
    'Debt_GDP', 'ExtDebt_GDP', 'DebtServ_XGDP', 'RealGDP_growth', 'GDP_per_capita_USD',
    'Inflation', 'fiscal_balance', 'primary_balance', 'CurrentAccount_GDP', 'Reserves_months',
    'ExchangeRate_change', 'Trade_openness', 'US_FedFundsRate', 'World_GDP_growth',
    'Oil_price_index', 'VIX', 'ICRG_political', 'ElectionYear'
]

X = df[features]
y = df['Default']

# 3️⃣ Define K-Fold
kf = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)

accuracies = []
f1_scores = []
feature_importances = np.zeros(len(features))

# 4️⃣ Cross-validation loop
for fold, (train_idx, test_idx) in enumerate(kf.split(X, y), 1):
    print(f"\n--- Fold {fold} ---")
    X_train, X_test = X.iloc[train_idx], X.iloc[test_idx]
    y_train, y_test = y.iloc[train_idx], y.iloc[test_idx]
    
    # 4️⃣ Apply SMOTE on training fold
    sm = SMOTE(random_state=42)
    X_train_res, y_train_res = sm.fit_resample(X_train, y_train)
    
    # 5️⃣ XGBoost model
    xgb_model = XGBClassifier(
        use_label_encoder=False,
        eval_metric='logloss',
        max_depth=4,
        learning_rate=0.05,
        n_estimators=200,
        subsample=0.8,
        colsample_bytree=0.8,
        random_state=42
    )
    
    xgb_model.fit(X_train_res, y_train_res)
    
    # 6️⃣ Predictions with threshold adjustment
    y_prob = xgb_model.predict_proba(X_test)[:, 1]
    threshold = 0.3
    y_pred = (y_prob >= threshold).astype(int)
    
    # 7️⃣ Metrics
    acc = accuracy_score(y_test, y_pred)
    f1 = f1_score(y_test, y_pred)
    accuracies.append(acc)
    f1_scores.append(f1)
    
    print(f"Accuracy: {acc:.3f}, F1 Score: {f1:.3f}")
    print(classification_report(y_test, y_pred))
    
    # 8️⃣ Accumulate feature importance
    feature_importances += xgb_model.feature_importances_

# 9️⃣ Average feature importance
feature_importances /= kf.get_n_splits()

# 🔟 Report cross-validation results
print("\n=== Cross-validation summary ===")
print(f"Mean Accuracy: {np.mean(accuracies):.3f} ± {np.std(accuracies):.3f}")
print(f"Mean F1 Score: {np.mean(f1_scores):.3f} ± {np.std(f1_scores):.3f}")

# 🔹 Feature importance ranking
importance_df = pd.DataFrame({
    'Feature': features,
    'Importance': feature_importances
}).sort_values(by='Importance', ascending=False)

print("\n=== Feature Importance ===")
print(importance_df)
